rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function profile() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    function role() {
      return signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)) ? profile().role : 'none';
    }
    function canRead() { return role() in ['admin', 'editor', 'viewer']; }
    function isAdmin() { return role() == 'admin'; }
    // Upload right for one data type: admins always; "Can upload" users only for the types ticked for them.
    function canUpload(k) {
      return isAdmin() || (role() == 'editor' && profile().get('perms', {}).get(k, false) == true);
    }

    // Settings (config/main) are admin-only. Hard Section months are stored as config/daily-YYYY-MM-<line>.
    match /config/{id} {
      allow read: if canRead();
      allow write: if isAdmin() || (id.matches('daily-.*') && canUpload('hard'));
    }

    // Future data types (e.g. forging plan vs achievement) live under modules/<type>/...
    match /modules/{mod}/{rest=**} {
      allow read: if canRead();
      allow write: if canUpload(mod);
    }

    // Old line-sheet data: readable, only the admin can change or delete it.
    match /entries/{id} { allow read: if canRead(); allow write: if isAdmin(); }
    match /targets/{id} { allow read: if canRead(); allow write: if isAdmin(); }

    // Who set the app up (first admin). Created once, never changed.
    match /meta/owner {
      allow read: if true;
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
    }

    // Last seen: each person can only write their own time and device; only the admin can read them.
    match /presence/{uid} {
      allow read: if isAdmin();
      allow write: if canRead() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['lastSeen', 'device', 'name']);
    }

    // People and their access. Only the admin can add, change or remove.
    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow create: if isAdmin() || (
        request.auth.uid == uid
        && request.resource.data.role == 'admin'
        && !exists(/databases/$(database)/documents/meta/owner)
        && getAfter(/databases/$(database)/documents/meta/owner).data.uid == uid);
      allow update, delete: if isAdmin();
    }
  }
}
